Debian 13:
Voraussetzung: es läuft kein tftp Server auf port 69
Ziel: Recover Xiaomi redme ax6s via tftp bootloader und Serial port
URL: https://openwrt.org/toh/xiaomi/ax3200
     https://github.com/mikeeq/xiaomi_ax3200_openwrt#uart-flash


> sudo su
> apt install tcpdump
> apt install dnsmasq

# tftp via dnsmasq
dnsmasq --no-daemon -i eth1 --dhcp-range=192.168.31.1,192.168.31.99 --enable-tftp --tftp-root=/srv/tftp --dhcp-boot=C0A81F01.img -p0 -K --log-dhcp --bootp-dynamic

# Verzeichnisrechte anpassen, damit man als normaler Nutzer die Firmware kopieren kann
> chmod 777 /srv/tftp

# Linux erwartete IP addresse geben
> ip addr add 192.168.31.100/24 dev eth0

# tftpboot Firmware kopieren. diese wird im uboot mit "tftpboot" nur in den ram geladen.
> cp freifunk-dresden-24.10.5-mediatek-mt7622-xiaomi_redmi-router-ax6s-initramfs-recovery.itb /srv/tftp/C0A81F01.img

# wenn recovery ram image geladen wurde:
# ubi-loader kopieren und flashen
> scp -O freifunk-dresden-24.10.5-mediatek-mt7622-xiaomi_redmi-router-ax6s-ubi-loader.itb root@192.168.1.1:/tmp/ubi.bin
